TABLE: products
CREATE TABLE products (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                barcode TEXT UNIQUE,
                name TEXT NOT NULL,
                price REAL NOT NULL,
                stock INTEGER NOT NULL DEFAULT 0
            , cost_price REAL DEFAULT 0.0, gst REAL, category_id INTEGER REFERENCES categories(id), tax_id INTEGER REFERENCES taxes(id), brand_id INTEGER REFERENCES brands(id), supplier_id INTEGER REFERENCES suppliers(id), has_variants BOOLEAN DEFAULT 0, attribute_set_id INTEGER REFERENCES attribute_sets(id))
---
TABLE: sqlite_sequence
CREATE TABLE sqlite_sequence(name,seq)

TABLE: sales
CREATE TABLE sales (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                total_amount REAL NOT NULL,
                payment_method TEXT NOT NULL,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
            , outlet_id INTEGER REFERENCES outlets(id), employee_id INTEGER REFERENCES employees(id), voided INTEGER DEFAULT 0, void_reason TEXT, customer_id INTEGER REFERENCES customers(id), discount_amount REAL DEFAULT 0, discount_type TEXT DEFAULT 'fixed', points_earned INTEGER DEFAULT 0, points_redeemed INTEGER DEFAULT 0)
---
TABLE: sale_items
CREATE TABLE sale_items (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                sale_id INTEGER,
                product_id INTEGER,
                quantity INTEGER NOT NULL,
                price_at_sale REAL NOT NULL, cost_price_at_sale REAL DEFAULT 0.0, variant_id INTEGER REFERENCES product_variants(id),
                FOREIGN KEY(sale_id) REFERENCES sales(id),
                FOREIGN KEY(product_id) REFERENCES products(id)
            )
---
TABLE: suppliers
CREATE TABLE suppliers (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        contact_person TEXT,
        phone TEXT,
        email TEXT,
        address TEXT
    )
---
TABLE: purchase_orders
CREATE TABLE purchase_orders (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        supplier_id INTEGER NOT NULL,
        status TEXT DEFAULT 'Draft',
        total_cost REAL DEFAULT 0.0,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY(supplier_id) REFERENCES suppliers(id)
    )
---
TABLE: purchase_order_items
CREATE TABLE purchase_order_items (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        po_id INTEGER NOT NULL,
        product_id INTEGER NOT NULL,
        quantity INTEGER NOT NULL,
        cost_price REAL NOT NULL,
        FOREIGN KEY(po_id) REFERENCES purchase_orders(id) ON DELETE CASCADE,
        FOREIGN KEY(product_id) REFERENCES products(id)
    )
---
TABLE: sale_payments
CREATE TABLE sale_payments (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            sale_id INTEGER,
            payment_method TEXT NOT NULL,
            amount REAL NOT NULL,
            FOREIGN KEY(sale_id) REFERENCES sales(id)
        )
---
TABLE: categories
CREATE TABLE categories (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            description TEXT
        )
---
TABLE: taxes
CREATE TABLE taxes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            rate REAL NOT NULL,
            is_default BOOLEAN DEFAULT 0
        )
---
TABLE: outlets
CREATE TABLE outlets (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            address TEXT,
            phone TEXT
        )
---
TABLE: employees
CREATE TABLE employees (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            role TEXT NOT NULL DEFAULT 'Cashier',
            pin_code TEXT UNIQUE NOT NULL
        , role_id INTEGER REFERENCES roles(id) DEFAULT 3)
---
TABLE: brands
CREATE TABLE brands (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            description TEXT
        )
---
TABLE: outlet_stock
CREATE TABLE outlet_stock (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            product_id INTEGER NOT NULL REFERENCES products(id),
            outlet_id INTEGER NOT NULL REFERENCES outlets(id),
            quantity INTEGER NOT NULL DEFAULT 0,
            UNIQUE(product_id, outlet_id)
        )
---
TABLE: stock_adjustments
CREATE TABLE stock_adjustments (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            product_id INTEGER NOT NULL REFERENCES products(id),
            outlet_id INTEGER NOT NULL REFERENCES outlets(id),
            employee_id INTEGER REFERENCES employees(id),
            quantity_change INTEGER NOT NULL,
            reason TEXT NOT NULL,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        )
---
TABLE: inventory_transfers
CREATE TABLE inventory_transfers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            product_id INTEGER NOT NULL REFERENCES products(id),
            from_outlet_id INTEGER NOT NULL REFERENCES outlets(id),
            to_outlet_id INTEGER NOT NULL REFERENCES outlets(id),
            quantity INTEGER NOT NULL,
            employee_id INTEGER REFERENCES employees(id),
            note TEXT,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        )
---
TABLE: cash_shifts
CREATE TABLE cash_shifts (
            id              INTEGER PRIMARY KEY AUTOINCREMENT,
            outlet_id       INTEGER NOT NULL REFERENCES outlets(id),
            employee_id     INTEGER REFERENCES employees(id),
            opening_float   REAL NOT NULL DEFAULT 0,
            closing_cash    REAL,
            expected_cash   REAL,
            variance        REAL,
            status          TEXT NOT NULL DEFAULT 'open',
            opened_at       DATETIME DEFAULT CURRENT_TIMESTAMP,
            closed_at       DATETIME,
            notes           TEXT
        )
---
TABLE: customers
CREATE TABLE customers (
            id         INTEGER PRIMARY KEY AUTOINCREMENT,
            name       TEXT NOT NULL,
            mobile     TEXT,
            email      TEXT,
            address    TEXT,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        , loyalty_points INTEGER DEFAULT 0)
---
TABLE: employee_attendance
CREATE TABLE employee_attendance (
      id          INTEGER PRIMARY KEY AUTOINCREMENT,
      employee_id INTEGER NOT NULL REFERENCES employees(id),
      outlet_id   INTEGER NOT NULL REFERENCES outlets(id),
      check_in    DATETIME NOT NULL DEFAULT (datetime('now','localtime')),
      check_out   DATETIME,
      hours       REAL,
      status      TEXT NOT NULL DEFAULT 'present',
      comment     TEXT
    )
---
TABLE: goods_received_notes
CREATE TABLE goods_received_notes (
      id            INTEGER PRIMARY KEY AUTOINCREMENT,
      grn_number    TEXT UNIQUE NOT NULL,
      po_id         INTEGER REFERENCES purchase_orders(id),
      outlet_id     INTEGER REFERENCES outlets(id),
      received_by   INTEGER REFERENCES employees(id),
      received_at   DATETIME NOT NULL DEFAULT (datetime('now','localtime')),
      notes         TEXT
    )
---
TABLE: grn_items
CREATE TABLE grn_items (
      id           INTEGER PRIMARY KEY AUTOINCREMENT,
      grn_id       INTEGER NOT NULL REFERENCES goods_received_notes(id),
      product_id   INTEGER NOT NULL REFERENCES products(id),
      qty_ordered  INTEGER NOT NULL DEFAULT 0,
      qty_received INTEGER NOT NULL DEFAULT 0,
      cost_price   REAL NOT NULL DEFAULT 0
    )
---
TABLE: returns
CREATE TABLE returns (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      original_sale_id INTEGER,
      outlet_id INTEGER NOT NULL REFERENCES outlets(id),
      employee_id INTEGER REFERENCES employees(id),
      customer_id INTEGER REFERENCES customers(id),
      total_refund REAL NOT NULL,
      refund_method TEXT NOT NULL,
      reason TEXT,
      timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
    )
---
TABLE: return_items
CREATE TABLE return_items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      return_id INTEGER NOT NULL REFERENCES returns(id),
      product_id INTEGER NOT NULL REFERENCES products(id),
      quantity INTEGER NOT NULL,
      refund_price REAL NOT NULL
    )
---
TABLE: roles
CREATE TABLE roles (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL UNIQUE,
      description TEXT,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )
---
TABLE: role_permissions
CREATE TABLE role_permissions (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      role_id INTEGER NOT NULL REFERENCES roles(id),
      permission TEXT NOT NULL,
      UNIQUE(role_id, permission)
    )
---
TABLE: attribute_sets
CREATE TABLE attribute_sets (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL UNIQUE,
      description TEXT
    )
---
TABLE: attributes
CREATE TABLE attributes (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      attribute_set_id INTEGER NOT NULL REFERENCES attribute_sets(id) ON DELETE CASCADE
    )
---
TABLE: attribute_options
CREATE TABLE attribute_options (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      attribute_id INTEGER NOT NULL REFERENCES attributes(id) ON DELETE CASCADE,
      value TEXT NOT NULL
    )
---
TABLE: product_variants
CREATE TABLE product_variants (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      product_id INTEGER NOT NULL REFERENCES products(id) ON DELETE CASCADE,
      sku TEXT UNIQUE NOT NULL,
      price REAL,
      cost_price REAL,
      stock INTEGER DEFAULT 0,
      timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
    )
---
TABLE: variant_option_values
CREATE TABLE variant_option_values (
      variant_id INTEGER NOT NULL REFERENCES product_variants(id) ON DELETE CASCADE,
      attribute_id INTEGER NOT NULL REFERENCES attributes(id) ON DELETE CASCADE,
      attribute_option_id INTEGER NOT NULL REFERENCES attribute_options(id) ON DELETE CASCADE,
      PRIMARY KEY (variant_id, attribute_id)
    )
---
TABLE: action_history
CREATE TABLE action_history (
            id          INTEGER PRIMARY KEY AUTOINCREMENT,
            user_name   TEXT,
            action_type TEXT NOT NULL,
            description TEXT NOT NULL,
            timestamp   DATETIME DEFAULT CURRENT_TIMESTAMP
        )
---
TABLE: supplier_returns
CREATE TABLE supplier_returns (
            id               INTEGER PRIMARY KEY AUTOINCREMENT,
            supplier_id      INTEGER REFERENCES suppliers(id),
            employee_id      INTEGER REFERENCES employees(id),
            outlet_id        INTEGER REFERENCES outlets(id),
            return_date      DATETIME DEFAULT CURRENT_TIMESTAMP,
            status           TEXT DEFAULT 'Pending',
            reason           TEXT,
            total_value      REAL DEFAULT 0,
            reference_number TEXT
        )
---
TABLE: supplier_return_items
CREATE TABLE supplier_return_items (
            id                INTEGER PRIMARY KEY AUTOINCREMENT,
            supplier_return_id INTEGER REFERENCES supplier_returns(id),
            product_id        INTEGER REFERENCES products(id),
            variant_id        INTEGER REFERENCES product_variants(id),
            quantity          INTEGER NOT NULL,
            unit_price        REAL,
            reason            TEXT
        )
